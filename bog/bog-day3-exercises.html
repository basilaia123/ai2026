import io

csv_data = """Item,Drawing_m2,Waste_Norm_Pct,Contractor_A_Bid_m2,Contractor_A_Rate_GEL
კერამოგრანიტის ფილა 60x120,142.0,8.0,172.0,80.0

კედლების შეღებვა Caparol,320.0,5.0,336.0,18.0"""

df = pd.read_csv(io.StringIO(csv_data))
df["Max_Allowed_m2"] = df["Drawing_m2"] * (1 + df["Waste_Norm_Pct"] / 100)
df["Excess_m2"] = (df["Contractor_A_Bid_m2"] - df["Max_Allowed_m2"]).apply(lambda x: max(0.0, round(x, 2)))
df["Excess_GEL"] = (df["Excess_m2"] * df["Contractor_A_Rate_GEL"]).round(2)
df["Status"] = df["Excess_m2"].apply(lambda x: "🚩 Red Flag" if x > 0 else "✓ Norm")
